# Oblivions 前端调试功能设计方案

> 版本：1.0 | 2026-06-12

---

## 一、目标

设计一个调试功能，当有人访问前端页面时，打印页面数据，让AI助手(Agents)可以看到前端页面到底是什么样子。

---

## 二、现状分析

### 2.1 已有调试基础设施

**移动调试日志** (`vex/js/map.js`):

- `moveDebugLog` 数组记录移动操作
- `moveDebugAdd(step, data)` 添加调试条目
- `renderMoveDebug()` 渲染到 `#moveDebugContent`
- `toggleMoveDebug()` 切换调试面板显示
- `flushDebugLog()` 上传到后端 `vex/cache/debug_move_{groomid}.log`

**后端日志接收** (`api_v2.php:458`):

- `handle_debug_log()` 接收前端调试数据
- 写入 `vex/cache/debug_move_{groomid}.log`

### 2.2 前端数据流

```
页面加载 → app.js:loadAll()
  ├── loadMap()      → api_v2.php?action=game_map
  ├── loadItemFind() → api_v2.php?action=player_info
  ├── loadExplorationMemory() → api_v2.php?action=player_info
  ├── loadInventory() → api_v2.php?action=player_inventory
  └── refreshLog()    → api_v2.php?action=game_log

移动操作 → clickMove() → submitCommand() → command.php
  └── 刷新: loadMap + loadPlayerInfo + loadItemFind + loadInventory + refreshLog
```

---

## 三、设计方案

### 3.1 设计原则

将调试功能独立为 `vex/js/debug.js`，与其他模块解耦，通过全局对象暴露接口。

### 3.2 核心功能

1. **通用调试框架**

   - `Debug.add(category, step, data)` 统一调试入口
   - 支持多个类别：API、地图、玩家、背包、操作日志
2. **调试面板 UI**

   - 固定在页面右下角的🐛按钮
   - 分标签显示不同类别的调试数据
   - 支持导出JSON文件
3. **自动记录**

   - 所有API调用自动记录响应数据
   - 移动操作自动记录完整流程
   - 后端日志按类别组织存储

### 3.3 使用方式

```html
<!-- 访问时加 ?debug=1 启用 -->
vex/index.html?debug=1
```

## 四、详细实现计划

### Phase 1：独立调试模块

**新增文件：`vex/js/debug.js`**

```javascript
// ══════════════════════════════════════════════════
// 调试模块 / Debug Module
// 独立的前端调试面板，记录所有API响应和操作数据
// ══════════════════════════════════════════════════

var Debug = (function() {
    // 调试类别
    var CATEGORIES = {
        API: 'api',
        MAP: 'map', 
        PLAYER: 'player',
        INVENTORY: 'inventory',
        LOG: 'log',
        ACTION: 'action'
    };
  
    // 调试日志存储
    var logs = {};
    var maxEntries = 50;
    var panelOpen = false;
    var currentTab = 'api';
    var enabled = false;
  
    // 初始化：检查URL参数
    function init() {
        var params = new URLSearchParams(window.location.search);
        if (params.get('debug') === '1' || params.get('debug') === 'auto') {
            enabled = true;
            createPanel();
            console.log('[Debug] 调试模式已启用');
        }
    }
  
    // 创建调试面板DOM
    function createPanel() {
        // 调试面板
        var panel = document.createElement('div');
        panel.id = 'debugPanel';
        panel.className = 'debug-panel';
        panel.style.display = 'none';
        panel.innerHTML = `
            <div class="debug-header">
                <h3>调试面板</h3>
                <div class="debug-tabs">
                    <button class="debug-tab active" data-tab="api">API</button>
                    <button class="debug-tab" data-tab="map">地图</button>
                    <button class="debug-tab" data-tab="player">玩家</button>
                    <button class="debug-tab" data-tab="inventory">背包</button>
                    <button class="debug-tab" data-tab="action">操作</button>
                </div>
                <button class="debug-export" onclick="Debug.exportData()">导出JSON</button>
                <button class="debug-close" onclick="Debug.togglePanel()">×</button>
            </div>
            <div class="debug-content" id="debugContent"></div>
        `;
        document.body.appendChild(panel);
      
        // 调试按钮
        var btn = document.createElement('button');
        btn.id = 'debugToggleBtn';
        btn.className = 'debug-toggle-btn';
        btn.title = '调试面板';
        btn.textContent = '🐛';
        btn.onclick = function() { Debug.togglePanel(); };
        document.body.appendChild(btn);
      
        // 绑定标签切换事件
        panel.querySelectorAll('.debug-tab').forEach(function(tab) {
            tab.onclick = function() { Debug.switchTab(this.dataset.tab); };
        });
    }
  
    // 添加调试记录
    function add(category, step, data) {
        if (!enabled) return;
        if (!logs[category]) logs[category] = [];
        logs[category].push({
            time: new Date().toISOString(),
            step: step,
            data: data
        });
        if (logs[category].length > maxEntries) {
            logs[category].shift();
        }
        if (panelOpen) renderPanel();
    }
  
    // 切换面板显示
    function togglePanel() {
        var panel = document.getElementById('debugPanel');
        var btn = document.getElementById('debugToggleBtn');
        if (!panel || !btn) return;
        panelOpen = !panelOpen;
        if (panelOpen) {
            panel.style.display = '';
            btn.textContent = '✕';
            renderPanel();
        } else {
            panel.style.display = 'none';
            btn.textContent = '🐛';
        }
    }
  
    // 切换标签
    function switchTab(tab) {
        currentTab = tab;
        document.querySelectorAll('.debug-tab').forEach(function(el) {
            el.classList.toggle('active', el.dataset.tab === tab);
        });
        renderPanel();
    }
  
    // 渲染面板内容
    function renderPanel() {
        var el = document.getElementById('debugContent');
        if (!el) return;
      
        var logsForTab = logs[currentTab] || [];
        if (logsForTab.length === 0) {
            el.innerHTML = '<p style="color:#666;">暂无数据</p>';
            return;
        }
      
        var html = '';
        for (var i = logsForTab.length - 1; i >= 0; i--) {
            var e = logsForTab[i];
            html += '<div class="debug-entry">' +
                '<div class="debug-step">[' + escapeHtml(e.step) + '] <span style="color:#666;">' + 
                escapeHtml(e.time.substring(11, 23)) + '</span></div>' +
                '<pre class="debug-data">' + escapeHtml(JSON.stringify(e.data, null, 2)) + '</pre>' +
                '</div>';
        }
        el.innerHTML = html;
    }
  
    // 导出调试数据
    function exportData() {
        var blob = new Blob([JSON.stringify(logs, null, 2)], {type: 'application/json'});
        var url = URL.createObjectURL(blob);
        var a = document.createElement('a');
        a.href = url;
        a.download = 'debug_' + new Date().toISOString().slice(0,19) + '.json';
        a.click();
    }
  
    // 获取所有调试数据
    function getData() {
        return logs;
    }
  
    // 上传调试日志到后端
    async function flush() {
        if (!enabled || Object.keys(logs).length === 0) return;
      
        var snapshot = {};
        for (var category in logs) {
            if (logs[category].length > 0) {
                snapshot[category] = logs[category].slice();
            }
        }
      
        if (Object.keys(snapshot).length === 0) return;
      
        try {
            await fetch(BASE_URL + '/api_v2.php?action=debug_log', {
                method: 'POST',
                credentials: 'include',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ categories: snapshot })
            });
        } catch (e) {
            console.error('[Debug] 日志上传失败:', e);
        }
    }
  
    // 公开API
    return {
        CATEGORIES: CATEGORIES,
        init: init,
        add: add,
        togglePanel: togglePanel,
        switchTab: switchTab,
        exportData: exportData,
        getData: getData,
        flush: flush
    };
})();

// 页面加载时初始化
Debug.init();
```

### Phase 3：后端日志增强

**修改文件：`api_v2.php`**

```php
function handle_debug_log() {
    global $groomid, $cuser;
  
    $raw = file_get_contents('php://input');
    $data = json_decode($raw, true);
    if (!$data || !is_array($data)) {
        api_error('无效的调试数据', 'INVALID_DEBUG_DATA');
    }
  
    // 按类别组织日志
    $categories = isset($data['categories']) ? $data['categories'] : [$data];
  
    foreach ($categories as $category => $entries) {
        $log_file = GAME_ROOT . "./vex/cache/debug_{$category}_{$groomid}.log";
        $dir = dirname($log_file);
        if (!is_dir($dir)) {
            mkdir($dir, 0777, true);
        }
      
        $entry = array(
            'time' => date('Y-m-d H:i:s'),
            'user' => $cuser,
            'entries' => $entries
        );
      
        $line = json_encode($entry, JSON_UNESCAPED_UNICODE) . "\n";
        file_put_contents($log_file, $line, FILE_APPEND | LOCK_EX);
    }
  
    api_response('success', array('written' => true));
}
```
