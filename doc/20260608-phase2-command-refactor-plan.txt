2.1 重构后 command.php 骨架

<?php
define('CURSCRIPT', 'game');
require './include/core/common.inc.php';
require GAME_ROOT.'./include/gamectl/game.func.php';
require GAME_ROOT.'./include/gamectl/player_auth.func.php';

// [A] 认证
$auth_result = auth_game_player();
// ... 5 种状态处理（不变）
$pdata = $auth_result['pdata'];

// [B] 公共初始化
require GAME_ROOT.'./include/gamectl/init_player.func.php';
extract($pdata, EXTR_REFS);
init_playerdata();
$log = init_player_log();
$cmd = $main = ''; $cmdnum = 0;
$gamedata = array();

// [C] 预执行检查
if ($hp > 0) {
    $log      .= init_noise_display();
    $rmcdtime  = init_cooldown();
    $log      .= init_dizzy_check();
    init_club_check();
    check_extrabag_overflow();

    // 预检查 + 路由分发
    $pre_check = resolve_pre_checks($hp, $command, $action, $mode,
        $coldtimeon, $rmcdtime, $sp_cmd);

    if ($pre_check['skip_cmd']) {
        $mode = $pre_check['mode'];
    } elseif ($pre_check['mode'] === 'chase_action') {
        // goto chase_flag 的消除
        include_once GAME_ROOT.'./include/game/combat/revbattle.func.php';
        \revbattle\revbattle_prepare($pre_check['command'], $message);
    } else {
        $mode = cmd_router_dispatch($command, $mode, $pdata, $cmdcdtime);
    }
}

// [D] 后处理
cmd_router_post_process($action, $gamestate, $bid,
    $coldtimeon, $cmdcdtime, $rmcdtime);
$endtime = $now;
$cmdnum++;

// [E] 响应组装
cmd_router_assemble_response($hp, $command, $mode, $state, $cmd,
    $itms0, $clbpara, $coldtimeon, $showcoldtimer, $rmcdtime,
    $pdata, $gamedata, $log, $main);

八、风险控制
================================================================

风险                                 概率   缓解措施
handler 中 require_once 顺序依赖      中     路由表统一加载，handler 内部不加 include
global 变量在 handler 函数间污染      中     每个 handler 显式声明所用 global
分支遗漏（elseif 未被路由表覆盖）    低     对照测试矩阵回归，else fallback 加 error_log
club21/club22 移动后功能异常         中     独立性强，先走通全流程再切
性能回退（函数调用开销）             低     include_once 次数减少，总体持平或更好
merge conflicts（新功能加入）        低     路由表是显式配置数组，加一行即可
